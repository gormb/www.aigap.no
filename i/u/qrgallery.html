-- 1) the "selected QR" field on the redir table (icon row)
--    token = <size><EC><hole>[t][u]   t = transparent, u = 21x uppercase host
--    e.g. 25M5  ·  25M5t  ·  21M5u  ·  21M5tu
--    (the old ^(21|25|29)[LMQH][03579]t?$ was WRONG: it could not record CAPS,
--     so it rejected 21M5u and made the 21x CAPS choice unrecordable)
alter table public.redir add column if not exists qr text;
comment on column public.redir.qr is
  'Selected QR variant: <size><EC><hole>[t][u]; t=transparent, u=21x uppercase host';
alter table public.redir drop constraint if exists redir_qr_fmt;
alter table public.redir add constraint redir_qr_fmt
  check (qr is null or qr ~ '^(21[LMQH][03579]t?u?|2[59][LMQH][03579]t?)$');

-- 2) ONLY if you want this page to record the choice from the browser.
--    Grants the public anon role UPDATE on `qr` so the picker can save.
--    NOTE: keep this at TABLE level. dbAdm.html shares the same anon key and
--    updates url/desc/group/sort, so `revoke update ... grant update (qr)`
--    breaks the admin page with 401 42501 permission denied for table redir.
create policy "redir update qr" on public.redir for update to anon
  using (true) with check (true);
grant update on public.redir to anon;
